{# Macro for creating standardized buttons #} {%- macro a_button(btn_cls, url, label) %} {%- set base_classes = "inline-flex items-center gap-2 px-4 py-2 text-sm font-medium rounded-md transition-colors focus:outline-none focus:ring-2 focus:ring-offset-2" %} {%- if btn_cls == 'primary' %} {%- set color_classes = "text-white bg-blue-600 hover:bg-blue-700 focus:ring-blue-500" %} {%- elif btn_cls == 'success' %} {%- set color_classes = "text-white bg-green-600 hover:bg-green-700 focus:ring-green-500" %} {%- elif btn_cls == 'danger' %} {%- set color_classes = "text-white bg-red-600 hover:bg-red-700 focus:ring-red-500" %} {%- elif btn_cls == 'warning' %} {%- set color_classes = "text-white bg-yellow-600 hover:bg-yellow-700 focus:ring-yellow-500" %} {%- else %} {%- set color_classes = "text-gray-700 bg-white border border-gray-300 hover:bg-gray-50 focus:ring-blue-500" %} {%- endif %} {{ label }} {%- endmacro %} {%- macro create_a(url, label) %} {{ a_button('primary', url, label) }} {%- endmacro %}
